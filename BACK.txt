/* =========================================================
   TOGGLE PROGRAM
========================================================= */

function toggleProgram(id){

    const program = document.getElementById("program" + id);

    if(!program){
        return;
    }

    program.classList.toggle("open");

}


/* =========================================================
   OPEN PROGRAM
========================================================= */

function openProgram(id){

    const program = document.getElementById("program" + id);

    if(!program){
        return;
    }

    document.querySelectorAll(".program").forEach(function(item){
        item.classList.remove("open");
    });

    program.classList.add("open");

    document.querySelectorAll(".side-item").forEach(function(item){
        item.classList.remove("active");
    });

    const sideItems = document.querySelectorAll(".side-item");

    sideItems.forEach(function(item){

        if(item.textContent.trim().startsWith(String(id))){
            item.classList.add("active");
        }

    });

    program.scrollIntoView({
        behavior:"smooth",
        block:"start"
    });

    if(window.innerWidth <= 900){
        document.getElementById("sidebar").classList.remove("show");
    }

}

   SEARCH

function searchPrograms(){

    const input =
        document.getElementById("searchInput").value
        .toLowerCase()
        .trim();

    document.querySelectorAll(".program").forEach(function(program){

        const text = program.innerText.toLowerCase();

        if(text.includes(input)){
            program.style.display = "";
        }
        else{
            program.style.display = "none";
        }

    });

}


/* =========================================================
   COPY CODE
========================================================= */

function copyCode(id){

    const data = programs[id];

    if(!data){
        return;
    }

    const code = data.code;

    if(navigator.clipboard){

        navigator.clipboard.writeText(code)
        .then(function(){

            showToast("Code copied!");

        })
        .catch(function(){

            fallbackCopy(code);

        });

    }
    else{

        fallbackCopy(code);

    }

}


/* =========================================================
   FALLBACK COPY
========================================================= */

function fallbackCopy(text){

    const textarea = document.createElement("textarea");

    textarea.value = text;

    textarea.style.position = "fixed";
    textarea.style.left = "-9999px";

    document.body.appendChild(textarea);

    textarea.focus();
    textarea.select();

    try{

        document.execCommand("copy");

        showToast("Code copied!");

    }
    catch(error){

        showToast("Copy failed");

    }

    document.body.removeChild(textarea);

}


/* =========================================================
   DOWNLOAD FILE
========================================================= */

function downloadFile(filename, content){

    const blob = new Blob(
        [content],
        {
            type:"text/plain;charset=utf-8"
        }
    );

    const url = URL.createObjectURL(blob);

    const link = document.createElement("a");

    link.href = url;
    link.download = filename;

    document.body.appendChild(link);

    link.click();

    document.body.removeChild(link);

    setTimeout(function(){

        URL.revokeObjectURL(url);

    },1000);

}

    }

    let content = "";

    content += "PYTHON PRACTICAL\n";
    content += "================\n\n";

    content += "PROGRAM " + id + "\n";
    content += data.name + "\n\n";

    content += "STEPS / ALGORITHM\n";
    content += "-----------------\n";

    data.steps.forEach(function(step,index){

        content += (index + 1) + ". " + step + "\n";

    });

    downloadFile(
        "program_" + id + "_steps.txt",
        content
    );

    showToast("Steps downloaded");

}

/* =========================================================
   MARK DONE
========================================================= */

const completed = new Set();


function markDone(id){

    const button = document.getElementById("done" + id);

    if(!button){
        return;
    }

    if(completed.has(id)){

        completed.delete(id);

        button.textContent = "✓ Mark Done";

        button.classList.remove("primary");

    }
    else{

        completed.add(id);

        button.textContent = "✓ Completed";

        button.classList.add("primary");

    }

    updateProgress();

}


/* =========================================================
   UPDATE PROGRESS
========================================================= */

function updateProgress(){

    const total = Object.keys(programs).length;

    const done = completed.size;

    const percentage =
        total === 0
        ? 0
        : Math.round((done / total) * 100);

    document.getElementById("progressText").textContent =
        done + " / " + total;

    document.getElementById("progressFill").style.width =
        percentage + "%";

}


/* =========================================================
   TOAST
========================================================= */

let toastTimer = null;


function showToast(message){

    const toast = document.getElementById("toast");

    toast.textContent = message;

    toast.classList.add("show");

    clearTimeout(toastTimer);

    toastTimer = setTimeout(function(){

        toast.classList.remove("show");

    },1800);

}
